cmake_minimum_required(VERSION 3.20)
project(wwhd_switch2_compiler LANGUAGES C CXX ASM)
get_filename_component(WWHD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT NINTENDO_SWITCH2)
  message(FATAL_ERROR "Use the isolated Switch 2 toolchain")
endif()
# libounce exposes a POSIX subset, not Win32. Select LLVM's Unix implementation
# explicitly; unsupported calls still need real port handling, not Linux claims.
set(UNIX TRUE)
add_compile_definitions(_GNU_SOURCE)
set(WWHD_LLVM_SOURCE "" CACHE PATH "Pinned llvm-project source checkout")
set(WWHD_NATIVE_TOOLS "" CACHE PATH "Host tablegen executables, NOT console executables")
if(NOT EXISTS "${WWHD_LLVM_SOURCE}/llvm/CMakeLists.txt")
  message(FATAL_ERROR "Missing pinned LLVM sources; run tools/switch2/build_compiler.py")
endif()
set(LLVM_TABLEGEN "${WWHD_NATIVE_TOOLS}/llvm-tblgen.exe" CACHE FILEPATH "Host LLVM tablegen" FORCE)
set(CLANG_TABLEGEN "${WWHD_NATIVE_TOOLS}/clang-tblgen.exe" CACHE FILEPATH "Host Clang tablegen" FORCE)
set(LLVM_NATIVE_TOOL_DIR "${WWHD_NATIVE_TOOLS}" CACHE PATH "Host build utilities" FORCE)
set(LLVM_ENABLE_PROJECTS "clang;lld" CACHE STRING "Integrated C frontend and ELF linker" FORCE)
set(LLVM_TARGETS_TO_BUILD "AArch64" CACHE STRING "Only the console code generator" FORCE)
set(LLVM_HOST_TRIPLE "aarch64-none-elf" CACHE STRING "Console-hosted LLVM" FORCE)
set(LLVM_DEFAULT_TARGET_TRIPLE "aarch64-none-elf" CACHE STRING "Generated guest object target" FORCE)
set(LLVM_ENABLE_PIC ON CACHE BOOL "" FORCE)
set(LLVM_ENABLE_THREADS ON CACHE BOOL "" FORCE)
foreach(feature ZLIB ZSTD LIBXML2 TERMINFO LIBEDIT FFI LIBPFM Z3 BACKTRACES CRASH_OVERRIDES)
  set(LLVM_ENABLE_${feature} OFF CACHE BOOL "No optional desktop dependency" FORCE)
endforeach()
foreach(component TESTS BENCHMARKS EXAMPLES DOCS)
  set(LLVM_INCLUDE_${component} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLVM_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CLANG_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CLANG_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_BINDINGS OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD 17)
include_directories(BEFORE "${CMAKE_CURRENT_LIST_DIR}/platform/include")
list(APPEND CMAKE_REQUIRED_INCLUDES "${CMAKE_CURRENT_LIST_DIR}/platform/include")
set(HAVE_SYSEXITS_H 1 CACHE INTERNAL "Provided by the console compiler port" FORCE)
add_subdirectory("${WWHD_LLVM_SOURCE}/llvm" llvm)
# Offline compilation needs no Unix-domain sockets (unsupported by Horizon).
get_target_property(support_sources LLVMSupport SOURCES)
list(FILTER support_sources EXCLUDE REGEX "(^|/)raw_socket_stream\\.cpp$")
set_property(TARGET LLVMSupport PROPERTY SOURCES "${support_sources}")
add_library(wwhd_compiler_platform STATIC platform/mapping_unavailable.cpp)
add_library(wwhd_nro_pack STATIC nro_pack.cpp nro_validate.cpp)
target_include_directories(wwhd_nro_pack PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
add_library(wwhd_console_clang STATIC console_compile.cpp console_link.cpp)
target_include_directories(wwhd_console_clang PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${WWHD_ROOT}/runtime/src")
target_compile_options(wwhd_console_clang PRIVATE -fno-rtti -fsigned-char)
target_include_directories(wwhd_console_clang PRIVATE
  "${WWHD_LLVM_SOURCE}/llvm/include" "${CMAKE_CURRENT_BINARY_DIR}/llvm/include"
  "${WWHD_LLVM_SOURCE}/lld/include"
  "${WWHD_LLVM_SOURCE}/clang/include" "${CMAKE_CURRENT_BINARY_DIR}/llvm/tools/clang/include")
llvm_map_components_to_libnames(compiler_backend AArch64CodeGen AArch64AsmParser AArch64Desc AArch64Info Support)
target_link_libraries(wwhd_console_clang PUBLIC clangCodeGen clangFrontend clangDriver lldELF lldCommon ${compiler_backend} wwhd_compiler_platform)
add_executable(wwhd_compiler_link_check EXCLUDE_FROM_ALL link_check.cpp)
set_target_properties(wwhd_compiler_link_check PROPERTIES SUFFIX ".elf")
target_link_libraries(wwhd_compiler_link_check PRIVATE wwhd_console_clang wwhd_nro_pack)
target_link_options(wwhd_compiler_link_check PRIVATE -Wl,--no-undefined)
option(WWHD_COMPILER_PYTHON "Link the actual console CPython RPX frontend" OFF)
if(WWHD_COMPILER_PYTHON)
  set(python_build "${WWHD_ROOT}/build/switch2-python")
  set(python_source "${WWHD_ROOT}/build/switch2-python-source")
  foreach(input libpython3.12.a Modules/_hacl/libHacl_Hash_SHA2.a Modules/expat/libexpat.a)
    if(NOT EXISTS "${python_build}/${input}")
      message(FATAL_ERROR "Build the actual Python library first: missing ${input}")
    endif()
  endforeach()
  find_library(WWHD_PYTHON_ZLIB z HINTS "${DEVKITPRO}/portlibs/switch/lib" NO_DEFAULT_PATH REQUIRED)
  add_library(wwhd_console_python STATIC console_python.cpp)
  target_include_directories(wwhd_console_python PUBLIC "${CMAKE_CURRENT_LIST_DIR}"
    PRIVATE "${python_build}" "${python_source}/Include")
  target_compile_options(wwhd_console_python PRIVATE -fsigned-char)
  target_link_libraries(wwhd_console_python PUBLIC "${python_build}/libpython3.12.a"
    "${python_build}/Modules/_hacl/libHacl_Hash_SHA2.a" "${python_build}/Modules/expat/libexpat.a"
    "${WWHD_PYTHON_ZLIB}")
  target_link_libraries(wwhd_compiler_link_check PRIVATE wwhd_console_python)
  target_compile_definitions(wwhd_compiler_link_check PRIVATE WWHD_COMPILER_PYTHON=1)
  add_library(wwhd_install_pipeline STATIC install_pipeline.cpp)
  set_target_properties(wwhd_install_pipeline PROPERTIES CXX_STANDARD 20)
  target_include_directories(wwhd_install_pipeline PRIVATE "${WWHD_ROOT}/runtime/src"
    "${WWHD_LLVM_SOURCE}/llvm/include" "${CMAKE_CURRENT_BINARY_DIR}/llvm/include")
  target_link_libraries(wwhd_install_pipeline PUBLIC wwhd_console_clang wwhd_console_python wwhd_nro_pack)
endif()
option(WWHD_BUILD_INSTALLER "Build the actual first-launch compilation NRO" OFF)
if(WWHD_BUILD_INSTALLER)
  include("${CMAKE_CURRENT_LIST_DIR}/Installer.cmake")
endif()
